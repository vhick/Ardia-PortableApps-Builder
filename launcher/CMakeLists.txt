cmake_minimum_required(VERSION 3.20)
project(ArdiaPortableLauncher LANGUAGES CXX RC)
add_executable(ArdiaPortable WIN32 main.cpp resource.rc)
target_compile_features(ArdiaPortable PRIVATE cxx_std_17)
target_compile_definitions(ArdiaPortable PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET ArdiaPortable PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(ArdiaPortable PRIVATE user32 kernel32)
if(MSVC)
  target_compile_options(ArdiaPortable PRIVATE /W4 /EHsc /utf-8)
  target_link_options(ArdiaPortable PRIVATE /MANIFEST:EMBED)
endif()
